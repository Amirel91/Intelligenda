    39→    39→const SA_TOKEN_KEY = 'superadmin_token'
    40→    40→
    41→    41→function getSuperAdminToken(): string | null {
    42→    42→  if (typeof window === 'undefined') return null
    43→    43→  return localStorage.getItem(SA_TOKEN_KEY)
    44→    44→}
    45→    45→
    46→    46→function clearSuperAdminToken() {
    47→    47→  if (typeof window === 'undefined') return
    48→    48→  localStorage.removeItem(SA_TOKEN_KEY)
    49→    49→}
    50→    50→
    51→    51→function authHeaders(): Record<string, string> {
    52→    52→  const token = getSuperAdminToken()
    53→    53→  return token
    54→    54→    ? { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' }
    55→    55→    : { 'Content-Type': 'application/json' }
    56→    56→}

    97→    97→const SUBSCRIPTION_LABELS: Record<string, { label: string; color: string; bg: string }> = {

   100→   100→  cancelling: { label: 'In disdetta', color: 'text-orange-700 dark:text-orange-400',  bg: 'bg-orange-50 dark:bg-orange-950/50' },
   101→   101→  suspended:  { label: 'Sospeso',     color: 'text-red-700 dark:text-red-400',     bg: 'bg-red-50 dark:bg-red-950/50' },
   102→   102→}
   103→   103→
   104→   104→function SubscriptionBadge({ status, planEndDate }: { status: string | null; planEndDate: string | null }) {
   105→   105→  const s = SUBSCRIPTION_LABELS[status || 'trial'] || SUBSCRIPTION_LABELS.trial
   106→   106→
   107→   107→  const isExpired = planEndDate && new Date(planEndDate) <= new Date()
   108→   108→  const effectiveStatus = isExpired && status === 'cancelling' ? 'suspended' : (status || 'trial')
   109→   109→  const config = SUBSCRIPTION_LABELS[effectiveStatus] || s
   110→   110→
   111→   111→  return (
   112→   112→    <div className="flex flex-col gap-0.5">
   113→   113→      <span className={`inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-xs font-medium w-fit ${config.bg} ${config.color}`}>
   114→   114→        <CreditCard className="w-3 h-3" />
   115→   115→        {config.label}
   116→   116→      </span>
   117→   117→      {planEndDate && (
   118→   118→        <span className="text-[10px] text-stone-400 dark:text-stone-500">
   119→   119→          {isExpired ? 'Scaduto il' : 'Rinnovo:'} {new Date(planEndDate).toLocaleDateString('it-IT', { day: '2-digit', month: 'short' })}
   120→   120→        </span>
   121→   121→      )}
   122→   122→    </div>
   123→   123→  )
   124→   124→}
