    37→// ==================== AUTH HELPER ====================
    38→
    39→const SA_TOKEN_KEY = 'superadmin_token'
    40→
    41→function getSuperAdminToken(): string | null {
    42→  if (typeof window === 'undefined') return null
    43→  return localStorage.getItem(SA_TOKEN_KEY)
    44→}
    45→
    46→function clearSuperAdminToken() {
    47→  if (typeof window === 'undefined') return
    48→  localStorage.removeItem(SA_TOKEN_KEY)
    49→}
    50→
    51→function authHeaders(): Record<string, string> {
    52→  const token = getSuperAdminToken()
    53→  return token
    54→    ? { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' }
    55→    : { 'Content-Type': 'application/json' }
    56→}

    95→// ==================== SUBSCRIPTION BADGE ====================

    97→const SUBSCRIPTION_LABELS: Record<string, { label: string; color: string; bg: string }> = {
    98→  trial:      { label: 'Prova',       color: 'text-blue-700 dark:text-blue-400',    bg: 'bg-blue-50 dark:bg-blue-950/50' },
    99→  active:     { label: 'Attivo',      color: 'text-emerald-700 dark:text-emerald-400', bg: 'bg-emerald-50 dark:bg-emerald-950/50' },
   100→  cancelling: { label: 'In disdetta', color: 'text-orange-700 dark:text-orange-400',  bg: 'bg-orange-50 dark:bg-orange-950/50' },
   101→  suspended:  { label: 'Sospeso',     color: 'text-red-700 dark:text-red-400',     bg: 'bg-red-50 dark:bg-red-950/50' },
   102→}
   103→
   104→function SubscriptionBadge({ status, planEndDate }: { status: string | null; planEndDate: string | null }) {
   105→  const s = SUBSCRIPTION_LABELS[status || 'trial'] || SUBSCRIPTION_LABELS.trial
   106→
   107→  const isExpired = planEndDate && new Date(planEndDate) <= new Date()
   108→  const effectiveStatus = isExpired && status === 'cancelling' ? 'suspended' : (status || 'trial')
   109→  const config = SUBSCRIPTION_LABELS[effectiveStatus] || s
   110→
   111→  return (
   112→    <div className="flex flex-col gap-0.5">
   113→      <span className={`inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-xs font-medium w-fit ${config.bg} ${config.color}`}>
   114→        <CreditCard className="w-3 h-3" />
   115→        {config.label}
   116→      </span>
   117→      {planEndDate && (
   118→        <span className="text-[10px] text-stone-400 dark:text-stone-500">
   119→          {isExpired ? 'Scaduto il' : 'Rinnovo:'} {new Date(planEndDate).toLocaleDateString('it-IT', { day: '2-digit', month: 'short' })}
   120→        </span>
   121→      )}
   122→    </div>
   123→  )
   124→}
